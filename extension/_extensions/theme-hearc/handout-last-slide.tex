% Version à distribuer : une page par frame, montrant l'état final des animations.
% L'option `handout` de Beamer affiche toutes les spécifications d'overlay
% simultanément (\only<1> et \only<2> superposés, overprint en erreur...).
% On compile donc en mode Beamer normal et on ne conserve, pour chaque frame,
% que sa dernière slide : chaque slide est mise de côté au shipout (la
% précédente est écartée), puis la dernière est émise en fin de frame.
% Les frames `allowframebreaks` ne passent pas par cette boucle et gardent
% toutes leurs pages.
\makeatletter
\newif\ifhearc@keeplast
\newbox\hearc@lastslide
\AddToHook{shipout/before}{%
  \ifhearc@keeplast
    \global\setbox\hearc@lastslide\copy\ShipoutBox
    \DiscardShipoutBox
    \global\advance\c@page\m@ne  % compensé par \@outputpage
  \fi}
\def\hearc@shiplastslide{%
  \global\hearc@keeplastfalse
  \ifvoid\hearc@lastslide\else
    \shipout\box\hearc@lastslide
    \global\advance\c@page\@ne
  \fi}
% \beamer@reseteecodes doit rester en dernier : il relance l'analyse du mode
% Beamer et absorbe ce qui le suit.
\pretocmd\beamer@doseveralframes{\global\hearc@keeplasttrue}{}
  {\PackageWarning{theme-hearc}{handout : \string\beamer@doseveralframes\space introuvable}}
\patchcmd\beamer@doseveralframes
  {\beamer@writeslidentry\beamer@reseteecodes}
  {\beamer@writeslidentry\hearc@shiplastslide\beamer@reseteecodes}{}
  {\PackageWarning{theme-hearc}{handout : \string\beamer@doseveralframes\space non modifiable}}
\makeatother
