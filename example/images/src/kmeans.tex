\documentclass[tikz,border=4pt]{standalone}
\definecolor{hearcred}{HTML}{FF0000}
\definecolor{hearcgray}{HTML}{6E6E6E}

% Points des deux groupes
\def\groupeA{0.8/1.0, 1.3/0.6, 1.0/1.6, 1.7/1.2, 0.5/0.4, 1.5/1.9, 2.0/0.7, 0.9/2.2}
\def\groupeB{3.6/2.8, 4.2/3.3, 3.9/2.2, 4.6/2.7, 3.3/3.4, 4.4/3.8, 3.7/3.9, 4.9/3.2}

% #1 : couleur du groupe A, #2 : couleur du groupe B, #3/#4 : centroïdes
\newcommand\etape[4]{%
  \begin{tikzpicture}[font=\sffamily\small]
    \draw[hearcgray!40] (0, 0) rectangle (5.5, 4.5);
    \foreach \x/\y in \groupeA \fill[#1] (\x, \y) circle (0.09);
    \foreach \x/\y in \groupeB \fill[#2] (\x, \y) circle (0.09);
    \node[centroide] at (#3) {};
    \node[centroide] at (#4) {};
  \end{tikzpicture}}

\usetikzlibrary{shapes.geometric}
\tikzset{centroide/.style={star, star points=5, minimum size=0.45cm, inner sep=0pt,
                           fill=white, draw=black, line width=0.9pt}}
\begin{document}
\etape{hearcgray}{hearcgray}{1.8,0.3}{3.0,4.2}  % 1. centroïdes initiaux
\etape{hearcred}{black}{1.8,0.3}{3.0,4.2}        % 2. affectation au plus proche
\etape{hearcred}{black}{1.2,1.2}{4.1,3.2}        % 3. recalcul des centroïdes
\end{document}
